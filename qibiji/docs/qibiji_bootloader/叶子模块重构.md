> 来源：`qibiji_boot/HARDWARE/cJSON/cJSON.c/.h`、`HARDWARE/MALLOC/malloc.c + mymalloc.h`、
> `HARDWARE/DS1302/ds1302.c/.h`。三个叶子都是 GPRS 链路的支撑件：mymalloc 提供 10KB 内存池、

## 2. mymalloc（原 `HARDWARE/MALLOC/malloc.c + mymalloc.h`）

### 2.1 角色与数据结构

定长块内存池：把 10KB 静态数组切成 32 字节一块，用一张 u16 管理表记录每块归属。

- `mem1base[10240]`：池本体，`__align(4)`（Keil 扩展，armcc 有效，保证对齐分配）；
- `mem1mapbase[320]`：管理表，`mem1mapbase[i]` = 第 i 块所属分配占的**连续块数**（0=空闲）；
- **分配方向**：`mem_malloc` 从表**末端向前**搜索连续空块——优先用池后段；
- 最小分配粒度 32 字节：`mymalloc(1)` 也占一整块，小块浪费率最高 31/32，但换来了 O(n) 简单表管理和零外部碎片。

### 2.2 行为要点

| 行为 | 说明 |
|---|---|
| 自动初始化 | `mem_malloc/mem_free` 发现 `memrdy=0` 时自动调 `mem_init()`——**main 不需要显式 mem_init()**（生产路径只有 cJSON 调 mymalloc，恰好覆盖） |
| `mymalloc` 失败 | 返回 NULL（池满或 size=0） |
| `myfree(NULL)` | 直接返回，安全 |
| `myrealloc` 失败 | 返回 NULL 且**旧块不释放**（内容仍有效），与标准 realloc 语义相容 |
| `myrealloc` 契约 | ptr 必须是 `mymalloc` 返回的池内地址且非 NULL（cJSON 用法天然满足：先 allocate 再 reallocate）；实现=新块+拷贝+释放旧块 |

### 2.3 改名与清理表

| 原名 | 处理 | 理由 |
|---|---|---|
| 文件 `malloc.c` | 改名 `mymalloc.c` | 与 `mymalloc.h` 对齐；未落盘前改名零成本（Keil 工程按新名添加） |
| `mem_init / mem_perused / mem_malloc / mem_free / mymemset / mymemcpy / mallco_dev` | **全部 static**，声明从头文件删除 | grep 证实全工程无外部消费者；`_m_mallco_dev` 结构体定义随之下沉到 .c |
| 对外 API | 只保留 `mymalloc / myfree / myrealloc` 三个声明 | cJSON 钩子契约面即此三个 |
| 注释「最大管理内存 8K」 | 修正为 10KB | `MEM1_MAX_SIZE=10*1024`，原注释与宏不符 |
| 注释「管理2个区域的内存」 | 修正为单池 | 本变体是单池裁剪版，`membase` 只有一项，注释是双池全版的残留 |
| 头文件 `#ifndef NULL` 块 | 删除 | `stm32f10x.h` 链已提供 NULL |

### 2.4 mymalloc.h（重写，全文）

```c
#ifndef __MYMALLOC_H
#define __MYMALLOC_H

#include "stm32f10x.h"   /* u8/u16/u32 基础类型 */

/* 内存池参数：10KB 池、32 字节块（芯片 STM32F103VE，RAM 64KB，池占用有余量） */
#define MEM1_BLOCK_SIZE        32                             /* 块大小 32 字节：最小分配粒度 */
#define MEM1_MAX_SIZE          10*1024                        /* 池总大小 10KB */
#define MEM1_ALLOC_TABLE_SIZE  MEM1_MAX_SIZE/MEM1_BLOCK_SIZE  /* 管理表项数 = 320 */

/* ---- 对外 API（内部实现全部 static，见 mymalloc.c） ---- */
void *mymalloc(u32 size);             /* 分配：返回池内地址；池满/size=0 返回 NULL */
void  myfree(void *ptr);              /* 释放：ptr 必须是 mymalloc 返回的地址；NULL 直接返回 */
void *myrealloc(void *ptr, u32 size); /* 重分配：新块+拷贝+释放旧块；失败返回 NULL 且旧块保留 */

#endif
```

### 2.5 mymalloc.c（重写，全文）

```c
/* mymalloc：定长块内存池（cJSON 的后端）。
   10KB 池按 32 字节分块，u16 管理表记录每块所属分配的连续块数；
   分配从表末端向前搜索连续空块。全部内部状态 static，对外只暴露三个 API。 */
#include "mymalloc.h"

/* ---- 池本体与管理表（static：全工程只经 mymalloc/myfree/myrealloc 间接访问） ---- */
__align(4) static u8  s_pool[MEM1_MAX_SIZE];              /* 内存池，4 字节对齐（Keil 扩展） */
static u16 s_map[MEM1_ALLOC_TABLE_SIZE];                  /* 管理表：0=空闲，否则=该分配的连续块数 */

/* ---- 内存管理控制器 ---- */
typedef struct {
    u8  ready;        /* 初始化完成标志：0=未就绪（首次分配/释放时自动 init） */
} mem_dev_t;
static mem_dev_t s_dev = { 0 };

/* 设内存 */
static void mymemset(void *s, u8 c, u32 count)
{
    u8 *xs = s;
    while (count--) *xs++ = c;
}

/* 复制内存 */
static void mymemcpy(void *des, void *src, u32 n)
{
    u8 *xdes = des;
    u8 *xsrc = src;
    while (n--) *xdes++ = *xsrc++;
}

/* 初始化：清管理表与池本体，置就绪标志 */
static void mem_init(void)
{
    mymemset(s_map, 0, MEM1_ALLOC_TABLE_SIZE * 2);   /* 表项是 u16，字节数 ×2 */
    mymemset(s_pool, 0, MEM1_MAX_SIZE);
    s_dev.ready = 1;
}

/* 内存分配（内部）：返回块偏移；失败返回 0xFFFFFFFF */
static u32 mem_malloc(u32 size)
{
    signed long offset;
    u16 nmemb;      /* 需要的连续块数 */
    u16 cmemb;      /* 当前连续空块计数 */
    u32 i;

    if (!s_dev.ready) mem_init();          /* 未初始化先初始化（生产路径不显式调 init） */

    if (size == 0) return 0xFFFFFFFF;
    nmemb = size / MEM1_BLOCK_SIZE;
    if (size % MEM1_BLOCK_SIZE) nmemb++;

    for (offset = MEM1_ALLOC_TABLE_SIZE - 1; offset >= 0; offset--)   /* 从池尾向前搜 */
    {
        if (!s_map[offset]) cmemb++;
        else                cmemb = 0;
        if (cmemb == nmemb)                /* 找到连续 nmemb 个空块 */
        {
            for (i = 0; i < nmemb; i++) s_map[offset + i] = nmemb;   /* 逐块标注所属分配 */
            return (u32)(offset * MEM1_BLOCK_SIZE);
        }
    }
    return 0xFFFFFFFF;                     /* 池满 */
}

/* 内存释放（内部）：0=成功 1=未就绪 2=偏移越界 */
static u8 mem_free(u32 offset)
{
    int i, index, nmemb;

    if (!s_dev.ready) { mem_init(); return 1; }

    if (offset < MEM1_MAX_SIZE)
    {
        index = offset / MEM1_BLOCK_SIZE;          /* 偏移所在块号 */
        nmemb = s_map[index];                      /* 该分配的连续块数 */
        for (i = 0; i < nmemb; i++) s_map[index + i] = 0;
        return 0;
    }
    return 2;
}

/* ---- 对外 API（mymalloc.h 声明） ---- */

/* 释放：ptr 必须是 mymalloc 返回的地址；NULL 直接返回 */
void myfree(void *ptr)
{
    u32 offset;
    if (ptr == NULL) return;
    offset = (u32)ptr - (u32)s_pool;
    mem_free(offset);                              /* 返回值丢弃：越界只发生在野指针场景 */
}

/* 分配：返回池内地址；池满/size=0 返回 NULL */
void *mymalloc(u32 size)
{
    u32 offset = mem_malloc(size);
    if (offset == 0xFFFFFFFF) return NULL;
    return (void *)((u32)s_pool + offset);
}

/* 重分配：新块+拷贝+释放旧块；失败返回 NULL 且旧块保留。
   契约：ptr 非 NULL 且为池内地址（cJSON 用法天然满足：先 allocate 再 reallocate）。
   注意与标准 realloc 的差异：myrealloc(NULL, n) 不成立（cJSON 不存在该调用形态）。 */
void *myrealloc(void *ptr, u32 size)
{
    u32 offset = mem_malloc(size);
    if (offset == 0xFFFFFFFF) return NULL;
    mymemcpy((void *)((u32)s_pool + offset), ptr, size);
    myfree(ptr);
    return (void *)((u32)s_pool + offset);
}
```

> 与原版的行为差异只有一处：原来 `mem_free` 在未就绪时返回 1，现在同样；`myfree` 的 err 返回值
> 原版就丢弃（printf 已注释），维持。其余逐行等价，只是 static 化 + 注释修正 + 池变量加 `s_` 前缀。

## 3. cJSON（第三方 1.7.19：原样入库 + realloc 钩子补丁）

### 3.1 错配问题（定案修订的理由）

cJSON 的三个内存钩子现状（cJSON.c:157-187，Keil 走非 MSVC 分支的宏替换）：

```c
#define internal_malloc  mymalloc    /* ✓ 已挂内存池 */
#define internal_free    myfree      /* ✓ 已挂内存池 */
#define internal_realloc realloc     /* ✗ 还是标准库！ */
```

**realloc 指向标准库而 malloc/free 指向池，是一颗已上膛的枪**：`cJSON_PrintUnformatted` 的
打印缓冲从 256 字节起、不够时走 `hooks->reallocate` 扩容（cJSON.c:1261-1263）——一旦有任何一次
打印超过 256 字节，就会**拿着内存池里的指针去调标准库 realloc**（对非堆指针 realloc 是未定义行为，
armcc 下轻则数据损坏、重则 HardFault）。当前工程没炸只是因为运气：唯一的活调用是 gprs_rx 的
Parse（只走 allocate/deallocate），print 路径全在注释掉的调试代码里。但这属于「谁先写一行
cJSON_Print 谁先踩」的雷，必须在本阶段拆掉。

**修法**：realloc 钩子改挂池版 `myrealloc`（§2.5 已确认其语义与 realloc 契约相容：新块+拷贝+
释放旧块、失败返回 NULL 旧块保留）。

### 3.2 补丁（共 4 行，其余 1300+ 行原样）

```c
/* cJSON.c:174-177（MSVC 分支） */
static void * CJSON_CDECL internal_realloc(void *pointer, size_t size)
{
    return myrealloc(pointer, size);      /* 原 realloc：标准库 realloc 对池指针是未定义行为 */
}

/* cJSON.c:181（非 MSVC 分支，Keil 走这里） */
#define internal_realloc myrealloc        /* 原 realloc */

/* cJSON.c:217（InitHooks 复位路径） */
global_hooks.reallocate = myrealloc;      /* 原 realloc */

/* cJSON.c:237（InitHooks 自定义路径的兜底） */
global_hooks.reallocate = myrealloc;      /* 原 realloc */
```

### 3.3 使用契约（记录，不在本阶段实施）

- `cJSON_Parse` 要求 **NUL 结尾**字符串——gprs_rx 的 `s_uart_rx2_json` 副本承担（JSON 尾后紧跟
  二进制帧数据，必须先截出副本）；上游 1.7.19 已提供 `cJSON_ParseWithLength`，可整体消掉这份
  副本拷贝，属 GPRS 阶段的候选优化，不在叶子阶段动；
- 解析树的释放必须 `cJSON_Delete` / 打印缓冲必须 `cJSON_free`（两者都经钩子走 `myfree`，
  不能与标准库 free 混用）。

## 4. DS1302（viewtool 驱动：迁移 + 清理）

### 4.1 角色与契约

RTC 日历时钟，gprs_tx 组报文时间戳的唯一数据源。**关键契约：`ds1302_dataread` 出口是十进制**
（内部已做 BCD→十进制转换，gprs_tx.c:19 注释依赖这一点）。引脚 PC8=CLK / PC7=DAT（开漏）/
PC6=RST，位翻转微秒级时序，无喂狗需求。

### 4.2 改名与清理表

| 原名 | 处理 | 理由 |
|---|---|---|
| 宏 `ds1302clk / ds1302dat / ds1302rst / ds1302gpio` | 改名 `DS1302_CLK / DS1302_DAT / DS1302_RST / DS1302_GPIO_PORT` | 小写宏与全工程大写宏风格不符；仅模块内 2 文件引用 |
| `uint8_t` | 统一改 `u8` | 与全工程类型风格一致（同 LCD 阶段处理） |
| `write_1302byte / read_1302 / write_1302 / ds1302_GPIO_Configuration` | **static 化**，从头文件删除声明 | grep 证实无外部调用方；函数名保留（描述准确） |
| `ds1302_settime` | **保留**（公开） | RTC 断电（VBAT 失效）后的补种工具，自带写-回读校验；唯一引用者 gprs_AT.c 是死文件，故本阶段无活调用，登记备查 |
| `ds1302_init` 里注释掉的播种块 + 死局部变量 `year / time[7]` | 删除 | 注释残留（编译器本就报 unused variable）；播种语义由 `ds1302_settime` 承载，需要时再显式调用 |
| 注释「Pc15配置为开漏模式」 | 修正 | 实际开漏配置在 PC7（DAT），PC15 从未被本文件触碰 |

### 4.3 ds1302_settime 的一个已知怪癖（保留原行为，登记）

校验回读时**跳过秒字段**（`for(i=1;i<7;i++)`）——秒在写入与回读之间必然走动，比它恒失败；
这是原驱动的正确设计，不是缺陷。settime 全部 7 字节都写、只验 1~6。

### 4.4 ds1302.h（重写，全文）

```c

#define _DS1302_H

#include <stm32f10x.h>

/* 引脚：PC8=CLK（推挽）、PC7=DAT（开漏，双向 IO）、PC6=RST（推挽） */

#define DS1302_RST        GPIO_Pin_6
#define DS1302_GPIO_PORT  GPIOC

/* ---- 对外 API ---- */

u8   ds1302_settime(u8 *dat);    /* 写 7 字节日历+回读校验（跳过秒位）：0=成功；断电补种工具 */
```

### 4.5 ds1302.c（重写，全文）

```c
/* DS1302 RTC 驱动（viewtool 原驱动，迁移清理版）。
   三线位翻转时序：CLK 上升沿写、下降沿读；DAT 开漏作双向 IO。
   ds1302_dataread 出口为十进制（gprs_tx 时间戳依赖此契约）。 */
#include "ds1302.h"

/* 秒/分/时/日/月/周/年 的寄存器读、写地址（BCD 格式） */
static const u8 s_reg_read[7]  = {0x81,0x83,0x85,0x87,0x89,0x8b,0x8d};
static const u8 s_reg_write[7] = {0x80,0x82,0x84,0x86,0x88,0x8a,0x8c};

/* IO 初始化：CLK/RST 推挽输出，DAT 开漏（可读可写） */
static void ds1302_GPIO_Configuration(void)
{
    GPIO_InitTypeDef GPIO_InitStruct;

    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOC, ENABLE);

    GPIO_InitStruct.GPIO_Pin   = DS1302_CLK | DS1302_RST;
    GPIO_InitStruct.GPIO_Mode  = GPIO_Mode_Out_PP;
    GPIO_InitStruct.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(DS1302_GPIO_PORT, &GPIO_InitStruct);

    GPIO_InitStruct.GPIO_Pin  = DS1302_DAT;
    GPIO_InitStruct.GPIO_Mode = GPIO_Mode_Out_OD;
    GPIO_Init(DS1302_GPIO_PORT, &GPIO_InitStruct);
}

/* 初始化：只配 GPIO。RTC 内容靠 VBAT 掉电保持，boot 不播种时间；
   需要补种（换电池后）时显式调 ds1302_settime */
void ds1302_init(void)
{
    ds1302_GPIO_Configuration();
}

/* 写一个字节：SCK 上升沿写数据，低位在前 */
static void write_1302byte(u8 dat)
{
    u8 i;
    for (i = 0; i < 8; i++)
    {
        GPIO_ResetBits(DS1302_GPIO_PORT, DS1302_CLK);
        if (dat & 0x01) GPIO_SetBits(DS1302_GPIO_PORT, DS1302_DAT);
        else            GPIO_ResetBits(DS1302_GPIO_PORT, DS1302_DAT);
        GPIO_SetBits(DS1302_GPIO_PORT, DS1302_CLK);
        dat >>= 1;
    }
}

/* 读一个寄存器：先写寄存器地址，再降沿读 8 位 */
static u8 read_1302(u8 add)
{
    u8 i, dat1 = 0x00;

    GPIO_ResetBits(DS1302_GPIO_PORT, DS1302_RST);
    GPIO_ResetBits(DS1302_GPIO_PORT, DS1302_CLK);
    GPIO_SetBits(DS1302_GPIO_PORT, DS1302_RST);       /* 拉高 RST 开始传输 */

    write_1302byte(add);
    for (i = 0; i < 8; i++)
    {
        GPIO_SetBits(DS1302_GPIO_PORT, DS1302_CLK);
        dat1 >>= 1;
        GPIO_ResetBits(DS1302_GPIO_PORT, DS1302_CLK); /* 下降沿后数据有效 */
        if (GPIO_ReadInputDataBit(DS1302_GPIO_PORT, DS1302_DAT) == 1)
            dat1 |= 0x80;
    }

    GPIO_ResetBits(DS1302_GPIO_PORT, DS1302_RST);     /* 释放总线 */
    return dat1;
}

/* 写一个寄存器 */
static void write_1302(u8 add, u8 dat)
{
    GPIO_ResetBits(DS1302_GPIO_PORT, DS1302_RST);
    GPIO_ResetBits(DS1302_GPIO_PORT, DS1302_CLK);
    GPIO_SetBits(DS1302_GPIO_PORT, DS1302_RST);

    write_1302byte(add);
    write_1302byte(dat);
    GPIO_ResetBits(DS1302_GPIO_PORT, DS1302_RST);
}

/* 读 7 字节日历并转十进制：ttt[0..6]=秒/分/时/日/月/周/年 */
void ds1302_dataread(char *ttt)
{
    u8 i;
    char x, y;
    for (i = 0; i < 7; i++)
    {
        ttt[i] = read_1302(s_reg_read[i]);
        x = ttt[i] % 16;        /* BCD 低 4 位 → 个位 */
        y = ttt[i] / 16 * 10;   /* BCD 高 4 位 → 十位 */
        ttt[i] = x + y;
    }
}

/* 写 7 字节日历+回读校验（跳过秒位：秒在写读之间必然走动，比它恒失败）。
   返回 0=校验一致；0xFF=任一字段不一致（RTC 电池失效/芯片故障的典型信号） */
u8 ds1302_settime(u8 *dat)
{
    u8 i, res = 0;
    char ttt[7];

    write_1302(0x8e, 0x00);                 /* 去写保护 */
    for (i = 0; i < 7; i++) write_1302(s_reg_write[i], dat[i]);
    write_1302(0x8e, 0x80);                 /* 加写保护 */

    ds1302_dataread(ttt);
    for (i = 1; i < 7; i++)
    {
        if (ttt[i] != dat[i]) { res = 0xFF; break; }
    }
    return res;
}
```

> 原版 `ds1302_settime` 校验循环里的 `res = 0` 赋值（每轮重复赋 0）合并为初值 0 + 不一致才置
> 0xFF，行为等价。其余逐行等价：宏改名、static 化、删被注释的 printf/延时。

## 5. 验证 main（替换当前 main，全文）

覆盖三个叶子的实测：池自动初始化 + 分配/释放/重分配/池满、cJSON 组包→打印（撑过 256B 初始
缓冲，**实机踩一遍 realloc 修正**）→解析回读、DS1302 读日历。不开狗（与 REG/USART 各阶段
验证 main 形态一致，所有延时 ≤0.5s）。

```c
/* USER/main.c —— 阶段 9 叶子验证：mymalloc / cJSON / DS1302 */
#include "stm32f10x.h"
#include "sys.h"
#include "delay.h"
#include "usart.h"
#include "led.h"
#include <stdio.h>      /* sprintf */
#include <string.h>     /* strlen/strcmp */
#include "mymalloc.h"
#include "cJSON.h"
#include "ds1302.h"

static u32 blink_cnt = 0;   /* 心跳计数 */

int main(void)
{
    u8    *p1, *p2, *p3;
    u8     i;
    char  *printed;        /* cJSON 打印缓冲（池内存，>256B 时走 realloc 修正路径） */
    cJSON *root, *parsed, *item;
    char   t[7];           /* DS1302 日历：秒/分/时/日/月/周/年（十进制） */

    delay_init();
    NVIC_Configuration();
    uart1_init(115200);
    LED_Init();
    LED_CLOSE();

    printf("leaf test start\r\n");

    /* ① 内存池：首次 mymalloc 触发自动 init（生产路径同款），分配/写满/读回 */
    p1 = mymalloc(100);
    if (p1 != NULL)
    {
        for (i = 0; i < 100; i++) p1[i] = i;          /* 写满 100 字节 */
        for (i = 0; i < 100; i++)
            if (p1[i] != i) break;
        printf("pool alloc/write: %s\r\n", (i == 100) ? "PASS" : "FAIL");
        myfree(p1);
    }
    else printf("pool alloc FAIL\r\n");

    /* 释放后再分配同尺寸 → 应复用刚释放的块（free 生效的证据） */
    p2 = mymalloc(100);
    printf("pool free/reuse: %s\r\n", (p2 == p1) ? "PASS" : "FAIL");
    p3 = mymalloc(MEM1_MAX_SIZE + 1);                 /* 超池请求 → NULL */
    printf("pool overflow:  %s\r\n", (p3 == NULL) ? "PASS" : "FAIL");

    /* ② myrealloc：扩容保内容 */
    p1 = myrealloc(p2, 300);
    if (p1 != NULL && p1[99] == 99) printf("pool realloc:   PASS\r\n");
    else                            printf("pool realloc:   FAIL\r\n");
    myfree(p1);

    /* ③ cJSON：组包 → 打印（>256B，实机踩 realloc 修正）→ 解析回读 → 释放 */
    root = cJSON_CreateObject();
    cJSON_AddStringToObject(root, "deviceKey", "dev001");
    cJSON_AddStringToObject(root, "productKey", "pk001");
    for (i = 0; i < 60; i++)                          /* 300 字节填充字段：逼出 realloc */
    {
        char tag[8];
        sprintf(tag, "pad%02u", (unsigned int)i);
        cJSON_AddStringToObject(root, tag, "0123456789012345");   /* 每项约 40B */
    }
    printed = cJSON_PrintUnformatted(root);           /* >256B → internal_realloc=myrealloc */
    if (printed != NULL)
    {
        printf("cjson print: %u bytes\r\n", (unsigned int)strlen(printed));
        parsed = cJSON_Parse(printed);                /* 解析路径（allocate） */
        item = (parsed != NULL) ? cJSON_GetObjectItem(parsed, "deviceKey") : NULL;
        printf("cjson roundtrip: %s\r\n",
               (item != NULL && strcmp(item->valuestring, "dev001") == 0) ? "PASS" : "FAIL");
        cJSON_Delete(parsed);
        cJSON_free(printed);                          /* 打印缓冲经钩子走 myfree，不能 free() */
    }
    else printf("cjson print FAIL\r\n");
    cJSON_Delete(root);

    /* ④ DS1302：读 7 字节日历打印（十进制，目检合法性；全 0 = RTC 从未写过，属正常） */
    ds1302_init();
    ds1302_dataread(t);
    printf("rtc: 20%02d-%02d-%02d %02d:%02d:%02d week%d\r\n",
           t[6], t[4], t[3], t[2], t[1], t[0], t[5]);

    while (1)
    {
        delay_ms(500);
        LED_ERR = !LED_ERR;
        blink_cnt++;
        printf("leaf alive, cnt=%u\r\n", (unsigned int)blink_cnt);
    }
}
```

**测试方法**：

1. 串口依次出现 `pool` 三行 PASS、`cjson print: N bytes`（N 应 > 1000，证明 realloc 修正路径
   真的跑过）、`cjson roundtrip: PASS`、`rtc: 20..-..-.. ..:..:..`；
2. `cjson print` 若为 FAIL 或 HardFault：先确认 cJSON.c 的 4 行钩子补丁（§3.2）已打——
   这正是补丁存在意义的反证实验，可把 `internal_realloc` 临时改回 `realloc` 复现炸机；
3. `rtc` 全 0 且 `ds1302_settime` 写读校验 0xFF → RTC 无 VBAT 供电或电池耗尽，登记现场处理。
